// vars
vec4 DiffuseColor;

// codes


void main() {

	// flow
	// code

	nodeVar2 = ( ( vec2( vec2( gl_FragCoord.xy.x, nodeUniform0.y - gl_FragCoord.xy.y ).x, ( nodeUniform0.y - vec2( gl_FragCoord.xy.x, nodeUniform0.y - gl_FragCoord.xy.y ).y ) ) - nodeVarying2 ).x * nodeVarying1.x );

		discard;
		

	}

	nodeVar5 = ( ( nodeVarying0 * nodeUniform2 ) * max( ( exp( ( nodeVar4 * -0.5 ) ) - nodeUniform3 ), 0.0 ) );

	if ( ( nodeVar5 < 0.0001 ) ) {

		discard;
		

	}

	nodeVar6 = clamp( nodeVar5, 0.0, 1.0 );
	DiffuseColor = vec4( nodeVarying3, nodeVarying4, nodeVar6, 1.0 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform5 );
	nodeVar7 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar7;
	gl_FragDepth = ( 1.0 - nodeVar6 );

	// result

}
