	// flow
	// code


	if ( ( nodeVarying1 <= 0.0 ) ) {

		discard;


	}

	nodeVar1 = dot( ( nodeVarying0 - vec2( 0.5 ) ), ( nodeVarying0 - vec2( 0.5 ) ) );

		discard;

	nodeVar5 = ( ( ( ( max( ( exp( ( pow( sqrt( ( nodeVar1 * 4.0 ) ), nodeVarying2 ) * -4.6051702 ) ) - 0.01 ), 0.0 ) * 1.0101010101010102 ) * nodeUniform2 ) * ( nodeVar4 * nodeVar4 ) ) * nodeVarying5 );

	if ( ( max( nodeVar3.x, max( nodeVar3.y, nodeVar3.z ) ) < 0.0001 ) ) {

		discard;


	}

	DiffuseColor = vec4( nodeVar3, nodeVar6 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform4 );
	nodeVar7 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar7;

	// result
	fragColor = nodeVar7;

}
