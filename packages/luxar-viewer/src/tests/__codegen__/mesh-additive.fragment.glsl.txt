// varyings

// codes

	// flow
	// code

	nodeVar1 = normalize( cross( dFdx( nodeVarying2 ), dFdy( nodeVarying2 ) ) );

	if ( ( nodeVar1.z < 0.0 ) ) {

		nodeVar2 = ( - nodeVar1 );

	} else {

		nodeVar2 = nodeVar1;

	}

	nodeVar3 = nodeVar2;
	nodeVar5 = dot( nodeVarying3, nodeVarying3 );

	if ( gl_FrontFacing ) {

		nodeVar6 = 1.0;

	} else {

		nodeVar6 = -1.0;

	}

	nodeVar7 = ( ( nodeVarying3 * vec3( inversesqrt( max( nodeVar5, 1e-12 ) ) ) ) * vec3( nodeVar6 ) );

	if ( ( nodeVar5 >= 1e-12 ) ) {

		nodeVar8 = nodeVar7;

	} else {

		nodeVar8 = nodeVar3;

	}

	nodeVar9 = nodeVar8;
	nodeVar10 = mix( nodeUniform0, 1.0, pow( clamp( ( ( nodeVar9.z * 0.5 ) + 0.5 ), 0.0, 1.0 ), nodeUniform1 ) );
	nodeVar12 = max( ( ( nodeVarying0 * vec3( nodeUniform2 ) ) + vec3( nodeUniform3 ) ), vec3( 0.0, 0.0, 0.0 ) );
	nodeVar13 = ( pow( nodeVar12, vec3( nodeUniform4 ) ) * vec3( nodeVar10 ) );
	nodeVar14 = ( nodeVarying1 * nodeUniform5 );
	DiffuseColor = vec4( nodeVar13, nodeVar14 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform6 );
	nodeVar15 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar15;

	// result
	fragColor = nodeVar15;

}
