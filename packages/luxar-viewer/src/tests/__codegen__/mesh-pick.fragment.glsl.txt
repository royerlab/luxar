// varyings
in float nodeVarying0;
flat  in float nodeVarying1;

// vars
vec4 DiffuseColor;

float nodeVar3;
float meshPickBrightness;

// codes

	// flow
	// code

	nodeVar0 = ( int( nodeUniform0 ) == 1 );
	meshPickCoverage = ( nodeVarying0 * nodeUniform1 );

	if ( ( nodeVar0 && ( meshPickCoverage < nodeUniform2 ) ) ) {

		discard;


	}


	if ( nodeVar0 ) {

		nodeVar3 = 1.0;

	} else {

		nodeVar3 = meshPickCoverage;

	}

	meshPickBrightness = clamp( nodeVar3, 0.0, 1.0 );

	if ( ( int( nodeUniform5 ) == 1 ) ) {

		nodeVar8 = ( cameraProjectionMatrixInverse * v_clipSpace );
		positionView = ( nodeVar8.xyz / vec3( nodeVar8.w ) );
		nodeVar5 = ( ( positionView.z + cameraNear ) / ( cameraNear - cameraFar ) );

	} else {

		nodeVar5 = ( 1.0 - meshPickBrightness );

	}

	gl_FragDepth = nodeVar5;

	// result
	fragColor = nodeVar4;

}
