// structs

// uniforms

	float nodeUniform0;

	float cameraNear;
	float cameraFar;
	mat4 cameraViewMatrix;
	mat4 cameraProjectionMatrix;
	mat4 cameraProjectionMatrixInverse;

// varyings
in float nodeVarying0;
in float nodeVarying1;
in vec4 v_clipSpace;
flat  in float nodeVarying3;
flat  in vec2 nodeVarying4;


// vars
float meshPickCoverage;
float meshPickNearFade;
float nodeVar0;

float nodeVar4;
float nodeVar5;

// codes


void main() {

	// flow
	// code

	meshPickCoverage = 0.0;
	meshPickCoverage = ( nodeVarying0 * nodeUniform0 );
	meshPickNearFade = 0.0;

	if ( ( cameraProjectionMatrix[ 3 ].w > 0.5 ) ) {

		nodeVar1 = 1;

	} else {

		nodeVar1 = 0;

	}


	if ( ( nodeVar1 == 1 ) ) {

		nodeVar0 = 1.0;

	} else {


		if ( ( nodeVarying1 >= 0.0 ) ) {

			nodeVar2 = 0.0;

		} else {

			nodeVar3 = max( nodeUniform2, 1e-20 );
			nodeVar2 = smoothstep( nodeVar3, ( nodeVar3 * 2.0 ), ( - nodeVarying1 ) );

		}

		nodeVar0 = nodeVar2;

	}

	meshPickNearFade = nodeVar0;
	meshPickCutout = false;
	meshPickCutout = ( int( nodeUniform3 ) == 1 );
	meshPickBrightness = 0.0;

	if ( meshPickCutout ) {

		nodeVar4 = 1.0;

	} else {

		nodeVar4 = meshPickCoverage;

	}

	meshPickBrightness = clamp( ( nodeVar4 * meshPickNearFade ), 0.0, 1.0 );

	if ( ( int( nodeUniform4 ) == 1 ) ) {

		nodeVar9 = ( cameraProjectionMatrixInverse * v_clipSpace );
		positionView = ( nodeVar9.xyz / vec3( nodeVar9.w ) );
		nodeVar5 = ( ( positionView.z + cameraNear ) / ( cameraNear - cameraFar ) );

	} else {

		nodeVar5 = ( 1.0 - meshPickBrightness );

	}

	gl_FragDepth = nodeVar5;

	if ( ( meshPickNearFade < 0.01 ) ) {

		discard;


	}


	if ( ( meshPickCutout && ( meshPickCoverage < nodeUniform11 ) ) ) {

		discard;


	}

	DiffuseColor = vec4( nodeVarying3, nodeVarying4.x, meshPickBrightness, nodeVarying4.y );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform13 );
	nodeVar12 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar12;

	// result
	fragColor = nodeVar12;

}
