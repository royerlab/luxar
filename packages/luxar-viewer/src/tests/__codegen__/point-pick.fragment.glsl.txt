precision highp float;

precision highp sampler3D;

precision highp usampler2DArray;

precision highp isampler2D;
precision highp isampler3D;

precision highp isampler2DArray;

precision highp sampler2DShadow;
precision highp sampler2DArrayShadow;
precision highp samplerCubeShadow;


// structs

layout( location = 0 ) out vec4 fragColor;



// uniforms

layout( std140 ) uniform object {
	float nodeUniform0;
	float nodeUniform1;
	float nodeUniform2;
	float nodeUniform3;
	mat4 nodeUniform5;
	float nodeUniform7;
	float nodeUniform8;
	vec2 nodeUniform9;
};


// varyings
in float nodeVarying0;
in vec2 nodeVarying1;
in float nodeVarying2;
in float nodeVarying3;
in float nodeVarying4;


// vars
vec4 DiffuseColor;
float nodeVar2;
float nodeVar4;
vec4 Output;
vec4 nodeVar5;

// codes


void main() {

	// flow
	// code


	if ( ( nodeVarying0 < 0.0001 ) ) {

		discard;
		

	}

	nodeVar2 = dot( ( nodeVarying1 - vec2( 0.5 ) ), ( nodeVarying1 - vec2( 0.5 ) ) );

	if ( ( nodeVar2 > 0.25 ) ) {

		discard;
		

	}

	nodeVar4 = ( max( ( exp( ( pow( sqrt( ( nodeVar2 * 4.0 ) ), nodeVarying2 ) * -4.6051702 ) ) - 0.01 ), 0.0 ) * 1.0101010101010102 );

	if ( ( nodeVar4 < 0.0001 ) ) {

		discard;
		

	}

	DiffuseColor = vec4( nodeVarying3, nodeVarying4, nodeVar4, 1.0 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform2 );
	nodeVar5 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar5;
	gl_FragDepth = ( 1.0 - clamp( nodeVar4, 0.0, 1.0 ) );

	// result
	fragColor = nodeVar5;

}
