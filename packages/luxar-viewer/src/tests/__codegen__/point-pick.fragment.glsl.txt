in float nodeVarying6;


// vars
vec4 DiffuseColor;

// codes


void main() {

	// flow
	// code


	if ( ( nodeVarying1 <= 0.0 ) ) {

		discard;
		

	}

	nodeVar1 = dot( ( nodeVarying0 - vec2( 0.5 ) ), ( nodeVarying0 - vec2( 0.5 ) ) );

	if ( ( nodeVar1 > 0.25 ) ) {

		discard;
		

	}

	nodeVar2 = ( ( ( max( ( exp( ( pow( sqrt( ( nodeVar1 * 4.0 ) ), nodeVarying2 ) * -4.6051702 ) ) - 0.01 ), 0.0 ) * 1.0101010101010102 ) * nodeVarying3 ) * pow( min( ( nodeVarying4 / 1.5 ), 1.0 ), 2.0 ) );

	if ( ( nodeVar2 < 0.0001 ) ) {

		discard;
		

	}

	DiffuseColor = vec4( nodeVarying5, nodeVarying6, nodeVar2, 1.0 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform4 );
	nodeVar4 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar4;
	gl_FragDepth = ( 1.0 - clamp( nodeVar2, 0.0, 1.0 ) );

	// result
	fragColor = nodeVar4;

}
