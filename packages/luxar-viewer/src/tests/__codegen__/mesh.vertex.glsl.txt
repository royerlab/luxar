// uniforms

layout( std140 ) uniform render {
	mat4 cameraViewMatrix;
	mat4 cameraProjectionMatrix;
};

layout( std140 ) uniform object {
	float nodeUniform0;
	float nodeUniform1;
	float nodeUniform2;
	float nodeUniform3;
	float nodeUniform4;
	float nodeUniform5;
	float nodeUniform6;
	float nodeUniform7;
	mat4 nodeUniform9;
	mat3 nodeUniform10;
};


// varyings
out vec3 nodeVarying0;
out float nodeVarying1;
out vec3 nodeVarying2;
out vec3 nodeVarying3;


// attributes
layout( location = 0 ) in vec3 position;
layout( location = 1 ) in vec4 color;
layout( location = 2 ) in vec3 normal;


// vars
vec3 nodeVar0;
vec3 nodeVar4;
vec3 nodeVar11;
mat4 modelViewMatrix;
vec4 nodeVar16;
float nodeVar17;
vec4 nodeVar18;

// codes


void main() {

	// transforms


	// flow
	// code

	nodeVar0 = vec3( 0.0, 0.0, 0.0 );
	nodeVarying2 = nodeVar0;
	nodeVar4 = vec3( 0.0, 0.0, 0.0 );
	nodeVarying3 = nodeVar4;
	nodeVar11 = vec3( 0.0, 0.0, 0.0 );
	nodeVarying0 = nodeVar11;
	nodeVarying1 = 1.0;
	modelViewMatrix = ( cameraViewMatrix * nodeUniform9 );
	nodeVar16 = ( modelViewMatrix * vec4( position, 1.0 ) );
	nodeVarying0 = color.xyz;

	if ( ( ( color.w < 1e30 ) && ( color.w > -1e30 ) ) ) {

		nodeVar17 = min( max( color.w, 0.0 ), 1.0 );

	} else {

		nodeVar17 = 1.0;

	}

	nodeVarying1 = nodeVar17;
	nodeVarying2 = nodeVar16.xyz;
	nodeVarying3 = ( cameraViewMatrix * vec4( ( nodeUniform10 * normal ), 0.0 ) ).xyz;
	nodeVar18 = ( cameraProjectionMatrix * nodeVar16 );

	// result
	gl_Position = nodeVar18;

	gl_PointSize = 1.0;

}
