	// transforms


	// flow
	// code

	nodeVarying1 = 0.0;

	nodeVar4 = ( int( uint( gl_VertexID ) ) / 65536 );

		nodeVar9 = 1.0;

	}

	nodeVarying0 = nodeVar9;
	modelViewMatrix = ( cameraViewMatrix * nodeUniform10 );
	nodeVar10 = ( modelViewMatrix * vec4( position, 1.0 ) );
	nodeVarying1 = nodeVar10.z;
	nodeVar11 = ( cameraProjectionMatrix * nodeVar10 );

	// result
	gl_Position = nodeVar11;
