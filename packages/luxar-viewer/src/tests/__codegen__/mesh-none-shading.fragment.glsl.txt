// varyings
in vec3 nodeVarying0;

vec3 nodeVar2;
float nodeVar3;
float nodeVar4;
float nodeVar5;
float nodeVar7;
float nodeVar8;

// codes


void main() {

	// flow
	// code

	nodeVar1 = max( ( ( nodeVarying0 * vec3( nodeUniform0 ) ) + vec3( nodeUniform1 ) ), vec3( 0.0, 0.0, 0.0 ) );
	nodeVar2 = pow( nodeVar1, vec3( nodeUniform2 ) );
	nodeVar3 = ( nodeVarying1 * nodeUniform3 );

	if ( ( int( nodeUniform4 ) == 1 ) ) {

		nodeVar4 = 1.0;

	} else {


		if ( ( nodeVarying2.z >= 0.0 ) ) {

			nodeVar5 = 0.0;

		} else {

			nodeVar7 = max( nodeUniform5, 1e-20 );
			nodeVar5 = smoothstep( nodeVar7, ( nodeVar7 * 2.0 ), ( - nodeVarying2.z ) );

		}

		nodeVar4 = nodeVar5;

	}

	nodeVar8 = nodeVar4;

	if ( ( nodeVar8 < 0.01 ) ) {

		discard;


	}


	if ( ( nodeVar3 < nodeUniform6 ) ) {

		discard;


	}

	DiffuseColor = vec4( ( nodeVar2 * vec3( nodeVar8 ) ), 1.0 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform7 );
	nodeVar9 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar9;

	// result
	fragColor = nodeVar9;

}
