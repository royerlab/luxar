precision highp sampler2DShadow;
precision highp sampler2DArrayShadow;
precision highp samplerCubeShadow;


// structs

layout( location = 0 ) out vec4 fragColor;



// uniforms

layout( std140 ) uniform object {
	mat4 nodeUniform1;
	float nodeUniform2;
	float nodeUniform3;
	float nodeUniform4;
	float nodeUniform5;
	float nodeUniform6;
	float nodeUniform7;
	float nodeUniform8;
	float nodeUniform9;
	float nodeUniform10;
	float nodeUniform12;
	vec2 nodeUniform13;
};


// varyings
in float nodeVarying0;
in vec2 nodeVarying1;
in vec3 nodeVarying2;
in float nodeVarying3;
in float nodeVarying4;
in float nodeVarying5;


// vars
vec4 DiffuseColor;
float nodeVar2;
vec3 nodeVar3;
float nodeVar6;
float nodeVar11;
vec4 Output;
vec4 nodeVar12;

// codes


void main() {

	// flow
	// code


	if ( ( nodeVarying0 < 0.0001 ) ) {

		discard;
		

	}

	nodeVar2 = dot( ( nodeVarying1 - vec2( 0.5 ) ), ( nodeVarying1 - vec2( 0.5 ) ) );

	if ( ( nodeVar2 > 0.25 ) ) {

		discard;
		

	}

	nodeVar3 = max( ( ( nodeVarying2 * vec3( nodeUniform3 ) ) + vec3( nodeUniform4 ) ), vec3( 0.0, 0.0, 0.0 ) );

	if ( ( max( nodeVar3.x, max( nodeVar3.y, nodeVar3.z ) ) < 0.0001 ) ) {

		discard;
		

	}

	nodeVar6 = min( ( nodeVarying4 / 1.5 ), 1.0 );
	nodeVar11 = ( ( ( ( max( ( exp( ( pow( sqrt( ( nodeVar2 * 4.0 ) ), nodeVarying3 ) * -4.6051702 ) ) - 0.01 ), 0.0 ) * 1.0101010101010102 ) * nodeUniform6 ) * ( nodeVar6 * nodeVar6 ) ) * nodeVarying5 );
	DiffuseColor = vec4( ( pow( nodeVar3, vec3( nodeUniform5 ) ) * vec3( nodeVar11 ) ), nodeVar11 );
	DiffuseColor.w = ( DiffuseColor.w * nodeUniform10 );
	nodeVar12 = max( vec4( DiffuseColor.xyz, DiffuseColor.w ), 0.0 );
	Output = nodeVar12;

	// result
	fragColor = nodeVar12;

}
