precision highp sampler2DArrayShadow;
precision highp samplerCubeShadow;


// structs





// uniforms

layout( std140 ) uniform render {
	mat4 cameraViewMatrix;
	mat4 cameraProjectionMatrix;
};

layout( std140 ) uniform object {
	float nodeUniform0;
	mat3 nodeUniform2;
	vec2 nodeUniform3;
	uint nodeUniform4;
	float nodeUniform5;
	float nodeUniform6;
	float nodeUniform7;
	float nodeUniform8;
	float nodeUniform9;
	float nodeUniform10;
	float nodeUniform11;
	float nodeUniform12;
	float nodeUniform14;
	float nodeUniform15;
	mat4 nodeUniform17;
	mat3 nodeUniform18;
};


// varyings
out vec3 nodeVarying0;
out float nodeVarying1;
out vec3 nodeVarying2;
out vec3 nodeVarying3;
out vec2 nodeVarying4;


// attributes
layout( location = 0 ) in vec3 position;
layout( location = 1 ) in vec4 color;
layout( location = 2 ) in vec3 normal;


// vars
vec3 nodeVar6;
vec3 nodeVar10;
vec3 nodeVar18;
mat4 modelViewMatrix;
vec4 nodeVar29;
float nodeVar30;
vec4 nodeVar31;

// codes


void main() {

	// transforms


	// flow
	// code

	nodeVarying4 = vec2( 0.0, 1.0 );
	nodeVar6 = vec3( 0.0, 0.0, 0.0 );
	nodeVarying2 = nodeVar6;
	nodeVar10 = vec3( 0.0, 0.0, 0.0 );
	nodeVarying3 = nodeVar10;
	nodeVar18 = vec3( 0.0, 0.0, 0.0 );
	nodeVarying0 = nodeVar18;
	nodeVarying1 = 1.0;
	modelViewMatrix = ( cameraViewMatrix * nodeUniform17 );
	nodeVar29 = ( modelViewMatrix * vec4( position, 1.0 ) );
	nodeVarying0 = color.xyz;

	if ( ( ( color.w < 1e30 ) && ( color.w > -1e30 ) ) ) {

		nodeVar30 = min( max( color.w, 0.0 ), 1.0 );

	} else {

		nodeVar30 = 1.0;

	}

	nodeVarying1 = nodeVar30;
	nodeVarying2 = nodeVar29.xyz;
	nodeVarying3 = ( cameraViewMatrix * vec4( ( nodeUniform18 * normal ), 0.0 ) ).xyz;
	nodeVar31 = ( cameraProjectionMatrix * nodeVar29 );
	nodeVarying4 = nodeVar31.zw;

	// result
	gl_Position = nodeVar31;

	gl_PointSize = 1.0;

}
